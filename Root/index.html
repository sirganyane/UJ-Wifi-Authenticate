// Function to be called in index.html to reflect tunnel traffic function registerTunnelConnection(id, type) { const stream = document.getElementById('monitor-stream'); const timestamp = new Date().toLocaleTimeString(); const ip = `10.22.${Math.floor(Math.random()*10)}.${Math.floor(Math.random()*254)}`; const row = ` ${timestamp} ${id} ${ip} GATEWAY_${type.toUpperCase()} CONNECTED `; stream.innerHTML = row + stream.innerHTML; }
